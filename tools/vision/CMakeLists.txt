cmake_minimum_required(VERSION 3.22)
project(RaInferVisionTools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_CPP_SOURCE "" CACHE PATH "Matching llama.cpp source checkout")
if(NOT EXISTS "${LLAMA_CPP_SOURCE}/tools/mtmd/mtmd.h")
  message(FATAL_ERROR "Set LLAMA_CPP_SOURCE to a Gemma4-capable llama.cpp checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_CPP_SOURCE}" llama EXCLUDE_FROM_ALL)
add_executable(rainfer_vision_encoder ../../src/vision_encoder.cpp)
target_include_directories(rainfer_vision_encoder PRIVATE "${LLAMA_CPP_SOURCE}/vendor" "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
if(MSVC)
  target_compile_options(rainfer_vision_encoder PRIVATE /utf-8)
endif()
target_link_libraries(rainfer_vision_encoder PRIVATE mtmd llama)
