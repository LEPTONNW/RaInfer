cmake_minimum_required(VERSION 3.22)
project(RaInfer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(mobile_dflash2 INTERFACE)
add_library(RaInfer::Core ALIAS mobile_dflash2)
target_include_directories(mobile_dflash2 INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(mobile_dflash2 INTERFACE Threads::Threads)

set(EXECUTORCH_INSTALL "" CACHE PATH "Pinned Android ExecuTorch CMake install prefix")
set(EXECUTORCH_SOURCE "" CACHE PATH "Matching ExecuTorch source checkout")
set(QNN_SDK_ROOT "$ENV{QNN_SDK_ROOT}" CACHE PATH "Locally licensed QAIRT 2.37.0.250724 SDK")
set(JSON_INCLUDE_DIR "" CACHE PATH "Directory containing nlohmann/json.hpp")
set(RAINFER_GLSLC "" CACHE FILEPATH "Vulkan shader compiler")
set(RAINFER_SPIRV_VAL "" CACHE FILEPATH "SPIR-V validator")

if(ANDROID)
  foreach(required EXECUTORCH_INSTALL EXECUTORCH_SOURCE QNN_SDK_ROOT)
    if(NOT IS_DIRECTORY "${${required}}")
      message(FATAL_ERROR "Set ${required} to an existing dependency directory. See docs/BUILD.md")
    endif()
  endforeach()
  if(NOT JSON_INCLUDE_DIR)
    set(JSON_INCLUDE_DIR "${EXECUTORCH_SOURCE}/extension/llm/tokenizers/third-party/json/single_include")
  endif()
  if(NOT EXISTS "${JSON_INCLUDE_DIR}/nlohmann/json.hpp")
    message(FATAL_ERROR "Set JSON_INCLUDE_DIR to a directory containing nlohmann/json.hpp")
  endif()
  if(CMAKE_HOST_WIN32)
    set(shader_host windows-x86_64)
    set(shader_suffix .exe)
  elseif(CMAKE_HOST_APPLE)
    set(shader_host darwin-x86_64)
    set(shader_suffix "")
  else()
    set(shader_host linux-x86_64)
    set(shader_suffix "")
  endif()
  if(NOT RAINFER_GLSLC)
    set(RAINFER_GLSLC "${ANDROID_NDK}/shader-tools/${shader_host}/glslc${shader_suffix}")
  endif()
  if(NOT RAINFER_SPIRV_VAL)
    set(RAINFER_SPIRV_VAL "${ANDROID_NDK}/shader-tools/${shader_host}/spirv-val${shader_suffix}")
  endif()
  foreach(shader_tool RAINFER_GLSLC RAINFER_SPIRV_VAL)
    if(NOT EXISTS "${${shader_tool}}")
      message(FATAL_ERROR "Set ${shader_tool} to an installed shader tool")
    endif()
  endforeach()
  find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
  set(shader_header "${CMAKE_CURRENT_BINARY_DIR}/generated/draft_head_spv.h")
  add_custom_command(OUTPUT "${shader_header}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/compile_draft_head.py"
      "${RAINFER_GLSLC}" "${RAINFER_SPIRV_VAL}"
      "${CMAKE_CURRENT_SOURCE_DIR}/src/draft_head.comp" "${shader_header}"
    DEPENDS src/draft_head.comp tools/compile_draft_head.py VERBATIM)
  add_custom_target(draft_head_shader DEPENDS "${shader_header}")
  list(PREPEND CMAKE_PREFIX_PATH "${EXECUTORCH_INSTALL}")
  find_package(absl CONFIG REQUIRED)
  find_package(executorch CONFIG REQUIRED)
  foreach(native_target mobile_runner pte_probe draft_probe target_capture)
    add_executable(${native_target} "src/${native_target}.cpp")
    target_sources(${native_target} PRIVATE src/vision_embedding.cpp)
    add_dependencies(${native_target} draft_head_shader)
    target_sources(${native_target} PRIVATE "${EXECUTORCH_SOURCE}/extension/llm/custom_ops/op_fallback.cpp")
    target_include_directories(${native_target} PRIVATE
      "${CMAKE_CURRENT_BINARY_DIR}/generated" "${JSON_INCLUDE_DIR}"
      "${EXECUTORCH_SOURCE}/.." "${EXECUTORCH_INSTALL}/schema/include"
      "${EXECUTORCH_SOURCE}/third-party/flatbuffers/include" "${QNN_SDK_ROOT}/include/QNN")
    target_compile_definitions(${native_target} PRIVATE ET_EVENT_TRACER_ENABLED)
    target_link_libraries(${native_target} PRIVATE mobile_dflash2 extension_module extension_tensor
      extension_data_loader extension_flat_tensor extension_named_data_map executorch executorch_core
      portable_ops_lib quantized_ops_lib portable_kernels quantized_kernels etdump flatccrt
      qnn_executorch_backend vulkan dl)
    set_target_properties(${native_target} PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
  endforeach()
  target_compile_options(mobile_runner PRIVATE -march=armv8.2-a+dotprod)
endif()

include(CTest)
if(BUILD_TESTING)
  foreach(test_name engine_contract_test argmax_test cpu_head_test hybrid_test image_rows_test preprocess_fixture cached_body_fixture)
    if(test_name STREQUAL "engine_contract_test")
      set(test_source engine_test)
    else()
      set(test_source ${test_name})
    endif()
    add_executable(${test_name} "tests/${test_source}.cpp")
    target_link_libraries(${test_name} PRIVATE mobile_dflash2)
  endforeach()
  add_test(NAME speculative_contract COMMAND engine_contract_test)
  add_test(NAME argmax_first_index_contract COMMAND argmax_test)
  add_test(NAME packed_cpu_head COMMAND cpu_head_test)
  add_test(NAME hybrid_pool_contract COMMAND hybrid_test)
  add_test(NAME image_row_injection_contract COMMAND image_rows_test)
endif()
